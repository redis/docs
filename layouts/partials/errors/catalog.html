{{- /* The error catalog (data/errors.json), ready for the pages that show it: the error-catalog shortcode, its Markdown expansion, and the error lookup. "url" $url) -}} {{- end -}} {{- $fix := cond (eq .module "json") "" (index $byPrefix (lower .prefix) | default "") -}} {{- if not $fix -}} {{- $text := lower .message -}} {{- range $byPhrase -}} {{- if not $fix -}} {{- if .err -}} {{- if hasPrefix $text (print "err " .phrase) -}}{{- $fix = .url -}}{{- end -}} {{- else if in $text .phrase -}} {{- $fix = .url -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $id := printf "e-%s" (substr (md5 (print .module " " .message)) 0 8) -}} {{- $group := cond (and (eq .module "timeseries") (hasPrefix .message "ERR TSDB:")) "ERR TSDB:" .prefix -}} {{- $entries = $entries | append (dict "id" $id "message" .message "explain" (.explain | default "") "prefix" .prefix "group" $group "module" .module "commands" $commands "list" (printf "%s#%s" (index $modules .module).url $id) "fix" $fix "verified" (.verified | default false)) -}} {{- end -}} {{- end -}} {{- return (dict "entries" $entries "modules" $modules) -}}
